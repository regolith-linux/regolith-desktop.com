{{- /* Image shortcode. Both forms work: {{< img "images/example.png" "Alt text" >}} {{< img src="images/example.png" alt="Alt text" >}} The path is looked up as a page resource (page bundles), then under assets/, then under static/. Large page and assets images are shown at 636 px wide, linked to the full-size file. A path that is found nowhere renders nothing and logs a warning (id img-missing). */ -}} {{- $src := "" -}} {{- $alt := "" -}} {{- if .IsNamedParams -}} {{- $src = .Get "src" | default "" -}} {{- $alt = .Get "alt" | default "" -}} {{- else -}} {{- $src = .Get 0 | default "" -}} {{- $alt = .Get 1 | default "" -}} {{- end -}} {{- $path := strings.TrimPrefix "/" $src -}} {{- $image := "" -}} {{- with .Page.Resources.GetMatch $path -}} {{- $image = . -}} {{- else -}} {{- with resources.Get $path -}} {{- $image = . -}} {{- end -}} {{- end -}} {{- if $image -}} {{- if and (in (slice "image/png" "image/jpeg" "image/gif" "image/webp") $image.MediaType.Type) (gt $image.Width 636) -}} {{- $preview := $image.Resize "636x" -}} {{- else -}} {{- end -}} {{- else if and $path (fileExists (path.Join "static" $path)) -}} {{- else -}} {{- warnidf "img-missing" "%s: image %q not found in the page bundle, assets/ or static/" .Page.File.Path $src -}} {{- end -}}
